    el.innerHTML=`<div class="bubble user">${mdLite(m.content)}</div>`;
  }else if(m.role==='subbot'){
    const sb=botById(m.sub_bot_id)||{name:m.sub_bot_id,icon:'team',color:'#999'};
    el=document.createElement('div');
    el.className='subbot-card';
    el.innerHTML=`<div class="subbot-card-head">${renderBotIconHTML(sb,22)}<span class="subbot-card-name">${esc(sb.name)}</span><span class="subbot-card-badge">✓ تم</span></div><div class="subbot-card-body">${mdLite(m.content)}</div>`;
  }else{
    el=document.createElement('div');
    el.className='msg-row bot';
    el.innerHTML=`<div class="bubble bot">${mdLite(m.content)}</div>`;
  }
  wrap.appendChild(el);
  wrap.scrollTop=wrap.scrollHeight;
  if(m.content)histCache[activeBotId+'_preview']=m.content.slice(0,60);
}

function showTyping(show){
  let t=document.getElementById('typingRow');
  if(show&&!t){
    t=document.createElement('div');
    t.id='typingRow';
    t.className='typing-row';
    t.innerHTML='<div class="typing-dots"><span></span><span></span><span></span></div><span>فريق العمل بيشتغل على طلبك...</span>';
    document.getElementById('messages').appendChild(t);
    document.getElementById('messages').scrollTop=document.getElementById('messages').scrollHeight;
  }else if(!show&&t){
    t.remove();
  }
}

async function selectBot(id){
  if(pollTimer){clearTimeout(pollTimer);pollTimer=null}
  activeBotId=id;
  document.querySelectorAll('.bot-row').forEach(r=>r.classList.toggle('active',r.dataset.bot===id));
  const bot=botById(id);
  updateHeaderAndDetails(bot);
  document.getElementById('msgInput').value='';
  setView('chat');
  document.getElementById('messages').innerHTML='<div style="text-align:center;color:var(--text3);font-size:12px;padding:20px">جاري تحميل المحادثة...</div>';
  try{
    const uid=getUid();
    const r=await fetch(`${POLL_URL}?user_id=${uid}&bot_id=${id}&since_id=0`,{headers:{'Referer':location.href}});
    const d=await r.json();
    sinceId=0;
    if(d.messages&&d.messages.length){
      renderMessages(d.messages);
      sinceId=d.messages[d.messages.length-1].id;
    }else{
      document.getElementById('messages').innerHTML=`<div style="text-align:center;color:var(--text3);font-size:12.5px;padding:30px 20px">${esc(bot.preview)}<br><br>اكتب أول رسالة وابدأ الشغل مع ${esc(bot.name)}</div>`;
    }
  }catch(e){
    document.getElementById('messages').innerHTML='<div style="text-align:center;color:var(--danger);font-size:12.5px;padding:20px">تعذر تحميل المحادثة، حاول تاني</div>';
  }
}

async function sendMessage(){
  if(sending)return;
  const input=document.getElementById('msgInput');
  const text=input.value.trim();
  if(!text)return;
  const uid=getUid();
  if(!uid){document.getElementById('loginGate').classList.add('show');return}
  sending=true;
  document.getElementById('sendBtn').disabled=true;
  input.value='';input.style.height='auto';
  appendMessageDOM({role:'user',content:text});
  showTyping(true);
  try{
    const resp=await fetch(GATEWAY,{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({bot_id:activeBotId,message:text,user_id:uid,channel_data:getChannelDataLS()})
    });
    const d=await resp.json();
    if(!d.ok){throw new Error('bad response')}
    sinceId=d.since_id;
    pollForReply();
  }catch(e){
    showTyping(false);
    appendMessageDOM({role:'assistant',content:'حصل خطأ في الاتصال، جرب تبعت تاني.'});
    sending=false;
    document.getElementById('sendBtn').disabled=false;
  }
}

function pollForReply(attempt){
  attempt=attempt||0;
  if(attempt>60){ // ~3 minutes max
    showTyping(false);
    appendMessageDOM({role:'assistant',content:'الطلب بياخد وقت أطول من المعتاد، جرب تفتح المحادثة تاني بعد شوية.'});
    sending=false;document.getElementById('sendBtn').disabled=false;
    return;
  }
  pollTimer=setTimeout(async()=>{
    try{
      const uid=getUid();
      const r=await fetch(`${POLL_URL}?user_id=${uid}&bot_id=${activeBotId}&since_id=${sinceId}`,{headers:{'Referer':location.href}});
      const d=await r.json();
      if(d.messages&&d.messages.length){
        d.messages.forEach(m=>{if(m.role!=='assistant'||!d.done)appendMessageDOM(m)});
        sinceId=d.messages[d.messages.length-1].id;
      }
      if(d.done){
        showTyping(false);
        const last=d.messages[d.messages.length-1];
        if(last&&last.role==='assistant')appendMessageDOM(last);
        sending=false;document.getElementById('sendBtn').disabled=false;
      }else{
        pollForReply(attempt+1);
      }
    }catch(e){
      pollForReply(attempt+1);
    }
  },2500);
}

  if(!getUid()){
    document.getElementById('loginGate').classList.add('show');
  }else{

  }
})();

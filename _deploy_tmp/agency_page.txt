async function sendMessage(){
  if(sending)return;
  const input=document.getElementById('msgInput');
  const text=input.value.trim();
  if(!text)return;
  const uid=getUid();
  if(!uid){document.getElementById('loginGate').classList.add('show');return}
  sending=true;
  document.getElementById('sendBtn').disabled=true;
  input.value='';input.style.height='auto';
  appendMessageDOM({role:'user',content:text});
  showTyping(true);
  try{
    const resp=await fetch(GATEWAY,{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({bot_id:activeBotId,message:text,user_id:uid,channel_data:getChannelDataLS()})
    });
    const d=await resp.json();
    if(!d.ok){throw new Error('bad response')}
    sinceId=d.since_id;
    pollForReply();
  }catch(e){
    showTyping(false);
    appendMessageDOM({role:'assistant',content:'حصل خطأ في الاتصال، جرب تبعت تاني.'});
    sending=false;
    document.getElementById('sendBtn').disabled=false;
  }
}

    return;
  }
  pollTimer=setTimeout(async()=>{
    try{
      const uid=getUid();
      const r=await fetch(`${POLL_URL}?user_id=${uid}&bot_id=${activeBotId}&since_id=${sinceId}`,{headers:{'Referer':location.href}});
      const d=await r.json();
      if(d.messages&&d.messages.length){
        d.messages.forEach(m=>{if(m.role!=='assistant'||!d.done)appendMessageDOM(m)});
        sinceId=d.messages[d.messages.length-1].id;
      }
      if(d.done){
        showTyping(false);
        const last=d.messages[d.messages.length-1];
        if(last&&last.role==='assistant')appendMessageDOM(last);
        sending=false;document.getElementById('sendBtn').disabled=false;
      }else{
        pollForReply(attempt+1);
      }
    }catch(e){
      pollForReply(attempt+1);
    }
  },2500);
}

  buildSidebar();
  const bot=botById(activeBotId);
  updateHeaderAndDetails(bot);
  if(!getUid()){
    document.getElementById('loginGate').classList.add('show');
  }else{
    selectBot(activeBotId);
  }
})();
